import React, { useEffect, useMemo, useState } from 'react';
import {
  Alert,
  FlatList,
  KeyboardAvoidingView,
  Platform,
  Pressable,
  SafeAreaView,
  StyleSheet,
  Text,
  TextInput,
  View,
} from 'react-native';
import AsyncStorage from '@react-native-async-storage/async-storage';
import { StatusBar } from 'expo-status-bar';

// ---- Theme ----------------------------------------------------------------

// Paper-planner palette: warm stock, ink text, one pine accent.
const colors = {
  paper: '#F5F0E6',
  card: '#FBF8F1',
  rule: '#DDD5C4',
  ink: '#1E2822',
  muted: '#7D7767',
  pine: '#2F5D50',
  pineSoft: '#D9E6DF',
  clay: '#B5573B',
};

// Serif for the date heading; system sans for everything else.
const fonts = {
  display: Platform.select({ ios: 'Georgia', android: 'serif', default: 'serif' }),
  body: Platform.select({ ios: 'System', android: 'sans-serif', default: 'sans-serif' }),
};

const space = { xs: 6, sm: 10, md: 16, lg: 24 };
const radius = 10;

// ---- App ------------------------------------------------------------------

const STORAGE_KEY = 'planner.tasks.v1';

// Each task lives in one of three "when" buckets until it is marked done.
const WHEN = [
  { key: 'today', label: 'Today' },
  { key: 'week', label: 'This week' },
  { key: 'later', label: 'Later' },
];

const TABS = [...WHEN, { key: 'done', label: 'Done' }];

const emptyMessages = {
  today: 'Nothing planned for today. Add one below.',
  week: 'Nothing lined up for this week yet.',
  later: 'No longer-term items. Park ideas here.',
  done: 'Completed tasks show up here.',
};

export default function App() {
  const [tasks, setTasks] = useState([]);
  const [loaded, setLoaded] = useState(false);
  const [tab, setTab] = useState('today');
  const [draft, setDraft] = useState('');
  const [when, setWhen] = useState('today');

  // Load once on launch.
  useEffect(() => {
    AsyncStorage.getItem(STORAGE_KEY)
      .then((raw) => {
        if (raw) setTasks(JSON.parse(raw));
      })
      .catch(() => {})
      .finally(() => setLoaded(true));
  }, []);

  // Save on every change, but never before the initial load finishes.
  useEffect(() => {
    if (!loaded) return;
    AsyncStorage.setItem(STORAGE_KEY, JSON.stringify(tasks)).catch(() => {});
  }, [tasks, loaded]);

  const counts = useMemo(() => {
    const result = { today: 0, week: 0, later: 0, done: 0 };
    for (const t of tasks) {
      if (t.done) result.done += 1;
      else result[t.when] += 1;
    }
    return result;
  }, [tasks]);

  const visible = useMemo(
    () =>
      tasks
        .filter((t) => (tab === 'done' ? t.done : !t.done && t.when === tab))
        .sort((a, b) => a.createdAt - b.createdAt),
    [tasks, tab]
  );

  const addTask = () => {
    const title = draft.trim();
    if (!title) return;
    const now = Date.now();
    setTasks((prev) => [
      ...prev,
      { id: String(now), title, when, done: false, createdAt: now },
    ]);
    setDraft('');
  };

  const toggle = (id) =>
    setTasks((prev) => prev.map((t) => (t.id === id ? { ...t, done: !t.done } : t)));

  const confirmRemove = (task) =>
    Alert.alert('Delete this task?', task.title, [
      { text: 'Keep it', style: 'cancel' },
      {
        text: 'Delete',
        style: 'destructive',
        onPress: () => setTasks((prev) => prev.filter((t) => t.id !== task.id)),
      },
    ]);

  const dateLine = new Date().toLocaleDateString(undefined, {
    weekday: 'long',
    month: 'long',
    day: 'numeric',
  });

  const renderTask = ({ item }) => (
    <Pressable
      onPress={() => toggle(item.id)}
      onLongPress={() => confirmRemove(item)}
      delayLongPress={350}
      style={({ pressed }) => [styles.row, pressed && styles.rowPressed]}
      accessibilityRole="checkbox"
      accessibilityState={{ checked: item.done }}
      accessibilityLabel={item.title}
      accessibilityHint="Double tap to toggle. Long press to delete."
    >
      <View style={[styles.check, item.done && styles.checkDone]}>
        {item.done ? <Text style={styles.checkMark}>✓</Text> : null}
      </View>
      <Text style={[styles.rowText, item.done && styles.rowTextDone]}>{item.title}</Text>
    </Pressable>
  );

  return (
    <SafeAreaView style={styles.screen}>
      <StatusBar style="dark" />

      <View style={styles.header}>
        <Text style={styles.date}>{dateLine}</Text>
        <Text style={styles.summary}>
          {counts.today} open today · {counts.week} this week · {counts.later} later
        </Text>
      </View>

      <View style={styles.tabs}>
        {TABS.map((t) => {
          const active = tab === t.key;
          const n = counts[t.key];
          return (
            <Pressable
              key={t.key}
              onPress={() => setTab(t.key)}
              style={[styles.tab, active && styles.tabActive]}
              accessibilityRole="tab"
              accessibilityState={{ selected: active }}
            >
              <Text style={[styles.tabText, active && styles.tabTextActive]}>
                {t.label}
                {n > 0 ? ` ${n}` : ''}
              </Text>
            </Pressable>
          );
        })}
      </View>

      <FlatList
        data={visible}
        keyExtractor={(t) => t.id}
        renderItem={renderTask}
        contentContainerStyle={visible.length ? styles.listContent : styles.emptyContainer}
        ListEmptyComponent={
          loaded ? <Text style={styles.empty}>{emptyMessages[tab]}</Text> : null
        }
      />

      <KeyboardAvoidingView
        behavior={Platform.OS === 'ios' ? 'padding' : undefined}
        style={styles.composer}
      >
        <View style={styles.whenRow}>
          {WHEN.map((w) => {
            const active = when === w.key;
            return (
              <Pressable
                key={w.key}
                onPress={() => setWhen(w.key)}
                style={[styles.chip, active && styles.chipActive]}
                accessibilityRole="radio"
                accessibilityState={{ selected: active }}
              >
                <Text style={[styles.chipText, active && styles.chipTextActive]}>{w.label}</Text>
              </Pressable>
            );
          })}
        </View>
        <View style={styles.inputRow}>
          <TextInput
            value={draft}
            onChangeText={setDraft}
            onSubmitEditing={addTask}
            placeholder="What needs doing?"
            placeholderTextColor={colors.muted}
            style={styles.input}
            returnKeyType="done"
            accessibilityLabel="New task"
          />
          <Pressable
            onPress={addTask}
            disabled={!draft.trim()}
            style={[styles.addButton, !draft.trim() && styles.addButtonDisabled]}
            accessibilityRole="button"
            accessibilityLabel="Add task"
          >
            <Text style={styles.addText}>Add</Text>
          </Pressable>
        </View>
      </KeyboardAvoidingView>
    </SafeAreaView>
  );
}

const styles = StyleSheet.create({
  screen: { flex: 1, backgroundColor: colors.paper },
  header: { paddingHorizontal: space.lg, paddingTop: space.md, paddingBottom: space.sm },
  date: { fontFamily: fonts.display, fontSize: 30, color: colors.ink, letterSpacing: -0.3 },
  summary: { fontFamily: fonts.body, fontSize: 14, color: colors.muted, marginTop: space.xs },

  tabs: {
    flexDirection: 'row',
    marginHorizontal: space.md,
    marginTop: space.sm,
    borderBottomWidth: StyleSheet.hairlineWidth,
    borderBottomColor: colors.rule,
  },
  tab: { paddingVertical: space.sm, paddingHorizontal: space.sm, marginRight: space.xs },
  tabActive: { borderBottomWidth: 2, borderBottomColor: colors.pine, marginBottom: -StyleSheet.hairlineWidth },
  tabText: { fontFamily: fonts.body, fontSize: 14, color: colors.muted },
  tabTextActive: { color: colors.pine, fontWeight: '600' },

  listContent: { paddingHorizontal: space.md, paddingTop: space.sm, paddingBottom: space.md },
  emptyContainer: { flexGrow: 1, justifyContent: 'center', padding: space.lg },
  empty: {
    fontFamily: fonts.display,
    fontStyle: 'italic',
    fontSize: 17,
    color: colors.muted,
    textAlign: 'center',
    lineHeight: 24,
  },

  row: {
    flexDirection: 'row',
    alignItems: 'center',
    backgroundColor: colors.card,
    borderRadius: radius,
    paddingVertical: space.md,
    paddingHorizontal: space.md,
    marginBottom: space.xs,
    borderWidth: StyleSheet.hairlineWidth,
    borderColor: colors.rule,
  },
  rowPressed: { opacity: 0.7 },
  check: {
    width: 24,
    height: 24,
    borderRadius: 12,
    borderWidth: 1.5,
    borderColor: colors.pine,
    alignItems: 'center',
    justifyContent: 'center',
    marginRight: space.md,
  },
  checkDone: { backgroundColor: colors.pine },
  checkMark: { color: colors.card, fontSize: 13, fontWeight: '700' },
  rowText: { flex: 1, fontFamily: fonts.body, fontSize: 16, color: colors.ink },
  rowTextDone: { color: colors.muted, textDecorationLine: 'line-through' },

  composer: {
    borderTopWidth: StyleSheet.hairlineWidth,
    borderTopColor: colors.rule,
    backgroundColor: colors.card,
    paddingHorizontal: space.md,
    paddingTop: space.sm,
    paddingBottom: space.md,
  },
  whenRow: { flexDirection: 'row', marginBottom: space.sm },
  chip: {
    paddingVertical: 6,
    paddingHorizontal: 12,
    borderRadius: 16,
    borderWidth: StyleSheet.hairlineWidth,
    borderColor: colors.rule,
    marginRight: space.xs,
  },
  chipActive: { backgroundColor: colors.pineSoft, borderColor: colors.pine },
  chipText: { fontFamily: fonts.body, fontSize: 13, color: colors.muted },
  chipTextActive: { color: colors.pine, fontWeight: '600' },

  inputRow: { flexDirection: 'row', alignItems: 'center' },
  input: {
    flex: 1,
    height: 46,
    paddingHorizontal: space.md,
    borderRadius: radius,
    borderWidth: StyleSheet.hairlineWidth,
    borderColor: colors.rule,
    backgroundColor: colors.paper,
    fontFamily: fonts.body,
    fontSize: 16,
    color: colors.ink,
  },
  addButton: {
    height: 46,
    paddingHorizontal: space.md,
    marginLeft: space.sm,
    borderRadius: radius,
    backgroundColor: colors.pine,
    justifyContent: 'center',
  },
  addButtonDisabled: { opacity: 0.4 },
  addText: { color: colors.card, fontSize: 16, fontWeight: '600' },
});
